# Vertex Cover Exercise - Combinatorial Optimization
import networkx as nx
import matplotlib.pyplot as plt
from pulp import *

# ==================== بخش ب و و: ILP و LP Relaxation ====================
def solve_vertex_cover_ilp(G, relaxation=False):
    prob = LpProblem("Vertex_Cover", LpMinimize)
    x = {v: LpVariable(f"x_{v}", cat="Continuous" if relaxation else "Binary", lowBound=0, upBound=1)
         for v in G.nodes()}

    # Objective
    prob += lpSum(x[v] for v in G.nodes())

    # Constraints
    for u, v in G.edges():
        prob += x[u] + x[v] >= 1

    prob.solve(PULP_CBC_CMD(msg=0))

    # تصحیح: شرط rounding برابر با >= 0.5 (پوشش همواره معتبر)
    cover = [v for v in G.nodes() if value(x[v]) >= 0.5]
    obj_value = value(prob.objective)
    return obj_value, cover, {v: value(x[v]) for v in G.nodes()}

# ==================== بخش ج و د: الگوریتم حریصانه ====================
def greedy_vertex_cover(G):
    G_copy = G.copy()
    cover = []
    while G_copy.edges():
        u, v = next(iter(G_copy.edges()))
        cover.extend([u, v])
        # حذف رأس‌ها و یال‌های incident
        G_copy.remove_node(u)
        if v in G_copy:
            G_copy.remove_node(v)
    return len(set(cover)), set(cover)

# ==================== نمونه‌های گراف برای تست ====================
G1 = nx.Graph()
G1.add_edges_from([(0,1), (1,2), (2,0), (2,3)])

G2 = nx.complete_bipartite_graph(3, 3)

G3 = nx.cycle_graph(5)

graphs = {"G1": G1, "G2": G2, "G3": G3}

print("نتایج برای نمونه‌ها:\n")

for name, G in graphs.items():
    print(f"--- گراف {name} ---")
    print("تعداد رأس:", G.number_of_nodes(), "تعداد یال:", G.number_of_edges())

    # ILP
    ilp_val, ilp_cover, _ = solve_vertex_cover_ilp(G, relaxation=False)
    print(f"ILP (بهینه): اندازه = {ilp_val}, پوشش = {sorted(ilp_cover)}")

    # LP Relaxation
    lp_val, lp_rounded, lp_values = solve_vertex_cover_ilp(G, relaxation=True)
    print(f"LP Relaxation: مقدار = {lp_val:.2f}")
    print(f"   مقادیر fractional: { {v: round(lp_values[v],3) for v in lp_values} }")
    print(f"   بعد از rounding ساده (>=0.5): اندازه تقریبی = {len(lp_rounded)}")

    # Greedy
    greedy_size, greedy_cover = greedy_vertex_cover(G)
    print(f"Greedy: اندازه = {greedy_size}, پوشش = {sorted(greedy_cover)}")

    print("\n")

    pos = nx.spring_layout(G)
    nx.draw(G, pos, with_labels=True, node_color='lightblue')
    plt.title(f"Graph {name}")
    plt.show()

\section*{تعریف مسئله پوشش رأس (\en{Vertex Cover})}
در یک گراف بدون جهت \( G = (V, E) \)، یک پوشش رأس، زیرمجموعه‌ای از رئوس \( C \subseteq V \) است به طوری که هر یال \( e \in E \) حداقل به یک رأس در \( C \) متصل باشد. هدف، یافتن پوشش رأس با کمترین تعداد رأس است.
\[
\min |C| \quad \text{s.t.} \quad \forall (u,v)\in E: \; u\in C \text{ یا } v\in C
\]

\separator

\section{الف) مدل برنامه‌ریزی عدد صحیح (\en{ILP})}
برای هر رأس \( v \in V \) یک متغیر دودویی \( x_v \) تعریف می‌کنیم:
\[
x_v = \begin{cases}
1 & \text{اگر رأس \( v \) در پوشش انتخاب شود} \\
0 & \text{در غیر این صورت}
\end{cases}
\]

مدل \en{ILP} به صورت زیر است:
\begin{align}
\min \quad & \sum_{v \in V} x_v \label{eq:obj}\\
\text{s.t.} \quad & x_u + x_v \geq 1 \quad \forall (u,v) \in E \label{eq:cons}\\
& x_v \in \{0,1\} \quad \forall v \in V \label{eq:binary}
\end{align}

\separator

\section{ج) طراحی الگوریتم حریصانه}
الگوریتم حریصانه ساده‌ای که در اینجا استفاده شده است، در هر مرحله یک یال دلخواه را انتخاب کرده و هر دو رأس آن را به پوشش می‌افزاید. سپس آن رئوس و تمام یال‌های متصل به آنها را از گراف حذف می‌کند. این روند تا زمانی که گراف بدون یال شود ادامه می‌یابد.

\begin{center}
\fbox{\parbox{0.85\textwidth}{
\textbf{الگوریتم حریصانه برای پوشش رأس} \\
1. \( C \gets \emptyset \) \\
2. \textbf{تا زمانی که} \( E \neq \emptyset \) \textbf{انجام بده:} \\
\phantom{2.} \hspace{0.5cm} یک یال \( (u,v) \) را به دلخواه انتخاب کن \\
\phantom{2.} \hspace{0.5cm} \( C \gets C \cup \{u, v\} \) \\
\phantom{2.} \hspace{0.5cm} همه یال‌های مجاور به \( u \) یا \( v \) را حذف کن \\
3. \textbf{برگردان} \( C \)
}}
\end{center}

این الگوریتم از نظر پیچیدگی زمانی \( O(|E|) \) با پیاده‌سازی مناسب است و ضریب تقریب ۲ را تضمین می‌کند (یعنی اندازه پوشش یافت شده حداکثر دو برابر اندازه بهینه است).

\section{و) \en{Relax} کردن مدل (\en{LP Relaxation}) و مقایسه}
در مدل \en{LP relaxation}، شرط صحیح بودن متغیرها (\( x_v \in \{0,1\} \)) را با محدودیت \( 0 \leq x_v \leq 1 \) جایگزین می‌کنیم. بدین ترتیب مدل به یک برنامه‌ریزی خطی تبدیل می‌شود که می‌تواند جواب‌های کسری داشته باشد.
\[
\min \sum_{v \in V} x_v \quad \text{s.t.} \quad x_u+x_v \geq 1, \quad 0 \leq x_v \leq 1
\]

این مدل یک کران پایین (\en{lower bound}) برای جواب بهینه \en{ILP} فراهم می‌کند. نتایج \en{LP} و گرد کردن ساده با شرط \( x_v \ge 0.5 \) (که همواره یک پوشش معتبر می‌سازد) در جدول \ref{tab:lp} مطابق با خروجی کد ارائه شده است.

\begin{table}[h!]
\centering
\caption{نتایج \en{LP relaxation} و \en{rounding} با شرط \( \ge 0.5 \)}
\label{tab:lp}
\begin{tabular}{|c|c|c|c|c|}
\hline
\textbf{گراف} & \textbf{مقدار \en{LP}} & \textbf{مقادیر کسری \( x_v \)} & \textbf{پوشش \en{rounded}} & \textbf{اندازه \en{rounded}} \\
\hline
\en{G1} & 2.00 & همه 0.5 & \{0,1,2,3\} & 4 \\
\en{G2} & 3.00 & \{0,1,2:0,  3,4,5:1\} & \{3,4,5\} & 3 \\
\en{G3} & 2.50 & همه 0.5 & \{0,1,2,3,4\} & 5 \\
\hline
\end{tabular}
\end{table}

\textbf{تحلیل:}
\begin{itemize}
\item گرد کردن با آستانه \( \ge 0.5 \) همیشه یک پوشش رأس معتبر می‌دهد؛ زیرا برای هر یال \( (u,v) \) داریم \( x_u + x_v \ge 1 \)، بنابراین دست‌کم یکی از متغیرها بزرگ‌تر یا مساوی 0.5 است و در مجموعه انتخاب می‌شود.
\item این روش یک الگوریتم ۲-تقریبی است: اندازه پوشش گردشده \( \le 2 \times \text{Opt}_{\text{LP}} \le 2 \times \text{Opt}_{\text{ILP}} \). در مثال‌های \en{G1} و \en{G3}، با وجود کسری بودن جواب \en{LP}، پوشش‌های گردشده به ترتیب اندازه ۴ و ۵ دارند که نسبت به بهینه (۲ و ۳) تقریب‌های ۲ و ۱.۶۷ هستند و همچنان در محدوده‌ی ضمانت ۲ قرار می‌گیرند.
\item \textbf{\en{Integrality gap}}: شکاف بین مقدار \en{LP} و \en{ILP}. در \en{G3}، \en{ILP}=3 و \en{LP}=2.5، بنابراین \en{gap} برابر \( \frac{3}{2.5}=1.2 \) است. در بدترین حالت، این شکاف برای مسأله پوشش رأس به ۲ نزدیک می‌شود (مانند گراف‌های کامل \( K_n \) که \en{ILP}=n-1 و \en{LP}=n/2). این نشان می‌دهد که \en{LP relaxation} به تنهایی کران پایین خوبی فراهم می‌کند ولی برای رسیدن به جواب صحیح نیاز به گرد کردن دارد.
\end{itemize}

\section{تعریف مسأله تخصیص خطی (\en{Linear Assignment Problem})}

مسأله تخصیص خطی یکی از مشهورترین مسائل بهینه‌سازی ترکیبیاتی است. فرض کنید $n$ کار و $n$ ماشین (یا فرد) داریم. هزینه (یا سود) اختصاص کار $i$ به ماشین $j$ برابر با $c_{ij}$ است. هدف آن است که هر کار دقیقاً به یک ماشین و هر ماشین دقیقاً یک کار اختصاص یابد، به گونه‌ای که مجموع هزینه‌های تخصیص کمینه (یا بیشینه) گردد.

به بیان ریاضی، مجموعه $X$ از تخصیص‌های یک‌به‌یک میان دو مجموعه $I=\{1,\dots,n\}$ و $J=\{1,\dots,n\}$ به صورت جایگشت $\pi$ از $\{1,\dots,n\}$ مدل می‌شود. مسأله عبارت است از:
\[
\min_{\pi \in S_n} \sum_{i=1}^{n} c_{i,\pi(i)},
\]
که $S_n$ مجموعه همه جایگشت‌های $n$ عنصری است.

\separator
\newpage
\section{مدل‌سازی به صورت برنامه‌ریزی عدد صحیح}

برای هر زوج $(i,j)$ متغیر دودویی $x_{ij}$ تعریف می‌کنیم:
\[
x_{ij} = \begin{cases}
1 & \text{اگر کار $i$ به ماشین $j$ اختصاص یابد}\\
0 & \text{در غیر این صورت}
\end{cases}
\]

مدل برنامه‌ریزی عدد صحیح $0$-$1$ به شکل زیر است:
\begin{align}
\min \quad & \sum_{i=1}^{n}\sum_{j=1}^{n} c_{ij} x_{ij} \label{eq:obj}\\
\text{s.t.} \quad & \sum_{j=1}^{n} x_{ij} = 1, \qquad \forall i=1,\dots,n \label{eq:assign1}\\
& \sum_{i=1}^{n} x_{ij} = 1, \qquad \forall j=1,\dots,n \label{eq:assign2}\\
& x_{ij} \in \{0,1\}, \qquad \forall i,j \label{eq:binary}
\end{align}
محدودیت \eqref{eq:assign1} تضمین می‌کند هر کار دقیقاً یک ماشین دریافت کند و محدودیت \eqref{eq:assign2} تضمین می‌کند هر ماشین دقیقاً یک کار انجام دهد.

\section{پیچیدگی: آیا مسأله تخصیص خطی \en{NP-hard} است؟}

خیر، مسأله تخصیص خطی \textbf{\en{NP-hard} نیست} و در زمان چندجمله‌ای قابل حل است. دلایل اصلی به شرح زیر است:

\begin{enumerate}
\item \textbf{وجود الگوریتم‌های چندجمله‌ای}: مشهورترین الگوریتم، الگوریتم مجارستانی (\en{Hungarian algorithm}) است که این مسأله را در زمان $O(n^3)$ حل می‌کند. همچنین می‌توان آن را به عنوان یک نمونه خاص از مسأله کمترین هزینه جریان (\en{Min-Cost Flow}) با ظرفیت‌های $1$ و $n$ گره مبدأ و $n$ گره مقصد در نظر گرفت که برای آن نیز الگوریتم‌های چندجمله‌ای وجود دارد.

\item \textbf{یکپارچگی (\en{Integrality}) ماتریس قیود}: ماتریس ضرایب قیود \eqref{eq:assign1} و \eqref{eq:assign2} یک ماتریس \textbf{تمام‌اً یک‌پارچه} (\en{Totally Unimodular}) است. یکی از خواص مهم ماتریس‌های تمام‌اً یک‌پارچه آن است که رأس‌های چندوجهی $\{x \ge 0 : Ax = b\}$ برای بردار صحیح $b$، دارای مختصات صحیح هستند. در نتیجه، \textbf{برنامه‌ریزی خطی حاصل از relaxing} کردن شرط صحیح بودن متغیرها ($0 \le x_{ij} \le 1$) یک جواب بهینه با مقادیر $0$ یا $1$ تولید می‌کند. بدین ترتیب، مسأله تخصیص خطی را می‌توان با حل یک برنامه‌ریزی خطی (که در زمان چندجمله‌ای قابل حل است) دقیقاً حل کرد، بدون این‌که نیاز به حل یک \en{ILP} عمومی باشد.

\item \textbf{جایگاه در کلاس $P$}: وجود الگوریتم چندجمله‌ای (چه ترکیبیاتی مانند مجارستانی و چه مبتنی بر \en{LP}) ثابت می‌کند که مسأله تخصیص خطی به کلاس پیچیدگی $P$ تعلق دارد و نمی‌تواند \en{NP-hard} باشد، مگر آن‌که $P = NP$.
\end{enumerate}

بنابراین، مسأله تخصیص یک مثال برجسته از مسائلی است که \en{IP} آن با وجود متغیرهای دودویی، ذاتاً «آسان» است و سخت‌ی \en{NP-hardness} عمومی برنامه‌ریزی عدد صحیح را به ارث نمی‌برد.

\section{پرسش تحلیلی: رابطه یک نمونه حل‌شدنی چندجمله‌ای با \en{NP-hardness} عمومی}

\textbf{اگر یک مسأله \en{integer programming} داشته باشیم که جوابش در زمان چندجمله‌ای به‌دست می‌آید، آیا می‌توان نتیجه گرفت که برنامه‌ریزی عدد صحیح \en{NP-hard} نیست؟}

پاسخ: \textbf{خیر، این نتیجه‌گیری نادرست است.} 

برنامه‌ریزی عدد صحیح عمومی ($IP$) یک چارچوب مدل‌سازی بسیار گسترده است. \en{NP-hard} بودن آن به این معنا است که \textbf{بدترین حالت} مسأله (یعنی وجود حداقل یک خانواده از نمونه‌ها) نیازمند زمان نمایی است، مگر $P=NP$. این خاصیت با ارائه یک نمونه خاص که بتوان آن را در زمان چندجمله‌ای حل کرد، نقض نمی‌شود. به عبارت دیگر:

\begin{itemize}
\item \textbf{\en{NP-hardness} یک ویژگی کل کلاس مسائل است، نه یک نمونه خاص.} یک مسأله \en{NP-hard} می‌تواند زیرمسائل ساده (قابل حل در $P$) فراوانی داشته باشد. برای مثال:
\begin{itemize}
\item مسأله کوله‌پشتی صفر و یک \en{NP-hard} است، اما اگر وزن همه اشیا یکسان باشد با یک مرتب‌سازی ساده و انتخاب حریصانه حل می‌شود.
\item مسأله پوشش رأسی \en{NP-hard} است، اما برای گراف‌های دوبخشی در زمان چندجمله‌ای (از طریق تطابق ماکزیمم) حل می‌شود.
\item مسأله تخصیص خطی که در بالا بررسی شد خود یک زیرمسأله ساده از \en{IP} است که ماتریس ضرایب آن ساختار ویژه‌ای دارد.
\end{itemize}
\item \textbf{کلاس $P$ در برابر $NP$}: اگر برای یک \en{IP} مشخص الگوریتم چندجمله‌ای یافت شود، صرفاً نشان می‌دهد که آن محدودیت خاص از $IP$ در کلاس $P$ جای دارد، نه آن‌که $IP$ عمومی در $P$ باشد.
\item برای رد \en{NP-hard} بودن کل $IP$، باید ثابت کرد که \textbf{هر} نمونه از برنامه‌ریزی عدد صحیح (از جمله مسائل \en{NP-complete} مانند \en{SAT}, \en{3-SAT}, رنگ‌آمیزی گراف و ... که همگی قابل فرمول‌بندی به صورت $IP$ هستند) در زمان چندجمله‌ای حل می‌شود. این امری است که با شناخت کنونی علوم کامپیوتر معادل با اثبات $P=NP$ خواهد بود.
\end{itemize}

در نتیجه، ارائه یک الگوریتم سریع برای یک مدل $IP$ مشخص، کوچک‌ترین خدشه‌ای به \en{NP-hardness} کلی برنامه‌ریزی عدد صحیح وارد نمی‌کند. \en{NP-hardness} بر اساس وجود \textbf{حداقل یک} نمونه سخت تعریف می‌شود، و پیدا شدن نمونه‌های آسان تأثیری در آن ندارد.
