#!/usr/bin/env python3
"""Educational entry point for the cohesive-bin-packing project.

The production implementation lives in :mod:`cohesive_bin_packing`.  Keeping
this file as a thin entry point prevents the notebook, scripts, and report from
silently using divergent algorithm versions.
"""

from pathlib import Path

from cohesive_bin_packing import (
    contract_instance,
    first_fit_decreasing,
    load_instance,
    lower_bound,
    solve_exact_ilp,
    tabu_search,
)


def demo() -> None:
    root = Path(__file__).resolve().parent
    path = root / "instances" / "small" / "n=30_C=100_pairs=10.txt"
    instance = load_instance(path)
    contracted = contract_instance(instance)
    bound = lower_bound(contracted.weights, contracted.capacity)
    ffd = first_fit_decreasing(contracted.weights, contracted.capacity)
    exact = solve_exact_ilp(instance, time_limit=120)
    result = tabu_search(instance, max_iter=500, tabu_tenure=7, target=exact["num_bins"] or bound)

    print(f"Instance: {path.name}")
    print(f"Original items: {instance['n']} | contracted components: {len(contracted.components)}")
    print(f"Lower bound: {bound} | FFD: {len(ffd)}")
    print(f"Exact ILP: {exact['num_bins']} ({exact['status']})")
    print(f"Tabu: {result.num_bins} | valid={result.valid} | time={result.solve_time:.6f}s")


if __name__ == "__main__":
    demo()

\section*{تطابق مدل ILP + گراف انقباض با فرمالیسم مدل‌های ترکیبیاتی}

فرمالیسم ۴-تایی (که معمولاً شامل فضای نمونه‌ها، فضای جواب‌های موجه، تابع هدف و جهت بهینه‌سازی است) یک تعریف انتزاعی و تئوریک (Theoretical) از مسائل بهینه‌سازی ترکیبیاتی است. اما ترکیب یک گراف انقباض (Contraction Graph) با یک مدل برنامه‌ریزی خطی عدد صحیح (ILP)، در واقع نمایش جبری و عملیاتی (Algebraic/Operational) دقیقاً همان فرمالیسم است.

با توجه به منطق مقاله‌ی (Doron-Arad et al. 2020) و مفاهیم پایه‌ای بهینه‌سازی، این دو به دلایل زیر کاملاً بر هم منطبق هستند و مدل ILP پیاده‌سازی شده در این پروژه یک مدل ترکیبیاتی کامل محسوب می‌شود:

\subsection*{۱. گراف انقباض = بازتعریف هوشمندانه‌ی فضای نمونه‌ها (Instance Space)}
در مسائل ترکیبیاتی پیچیده (مثل مقاله مذکور که با تضادهای گروهی Clique-graph conflicts سروکار دارد، و یا این پروژه که دارای محدودیت‌های Cohesive یا Must-link است)، گنجاندن تمام محدودیت‌ها در فرمول‌های ریاضی باعث انفجار فضای حالت می‌شود. استفاده از گراف انقباض (پیدا کردن مولفه‌های همبند و تبدیل آن‌ها به یک ابرگره یا Super-node)، در واقع یک پیش‌پردازش ساختاری است که فضای مسئله را به یک فضای معادل اما کوچک‌تر و خوش‌رفتارتر نگاشت می‌کند. در اینجا بخشی از محدودیت‌های قطعی در ساختار داده حل شده‌اند تا مدل نهایی تمیزتر شود.

\subsection*{۲. محدودیت‌های ILP = تعریف صریح فضای موجه (Feasible Space / Polytope)}
در فرمالیسم ۴-تایی، به یک مجموعه از جواب‌های موجه ($F$) نیاز است. در ILP، این فضای موجه به کمک معادلات و نامعادلات خطی ساخته می‌شود (مثل اینکه هر مولفه منقبض‌شده فقط در یک سطل قرار گیرد و ظرفیت سطل نقض نشود).
اگر به مقاله (Theorem 3.11) رجوع کنیم، نویسندگان برای اثبات وجود یک جواب موجه برای آیتم‌های کوچک، دقیقاً یک فضای موجه یا پلی‌توپ (Polytope) را با استفاده از محدودیت‌های ریاضی و متغیرهای تصمیم تعریف می‌کنند. محدودیت‌های ILP دقیقاً همین کار را می‌کنند: آن‌ها مرزهای هندسی فضای شدنی (Feasible Space) را به زبان جبر خطی می‌سازند.

\subsection*{۳. تابع هدف ILP = تابع هدف در فرمالیسم (Objective Function)}
متغیرهایی مثل $y$ در ILP و کمینه‌سازی جمع آن‌ها، دقیقاً همان تابع نگاشتی است که به هر نقطه از فضای موجه، یک هزینه (Cost) اختصاص می‌دهد و سالور به دنبال مینیمم کردن آن در مرزهای همان Polytope است.

\subsection*{نتیجه‌گیری}
فرمالیسم ۴-تایی، زبان نظریه مجموعه‌ها برای تعریف مسئله است، در حالی که ILP به همراه گراف انقباض، زبان جبر خطی و گراف برای مدل‌سازی دقیقاً همان مسئله است. گراف انقباض متولی مدیریت قیود ساختاری (Structural Constraints) است تا مدل کوچک شود، و متغیرهای باینری و محدودیت‌های ILP دقیقاً مرزهای فضای شدنی (Feasible Space) را می‌سازند و تابع هدف روی آن‌ها بهینه‌سازی می‌شود (مشابه رویکرد تعریف Polytope در مقالاتی مثل Doron-Arad et al). در نتیجه، این یک نمایش کاملاً استاندارد و فرمال از یک مدل بهینه‌سازی ترکیبیاتی است.

# Cohesive Bin Packing

Final project for Combinatorial Optimization: exact and heuristic solution of one-dimensional bin packing with hard must-link constraints.

## Reproducible workflow

Python 3.12 is recommended.

```bash
python3 -m pip install -r requirements.txt
python3 -m unittest discover -s tests -v
python3 run_experiments_final.py
python3 run_robustness.py
python3 generate_report_artifacts.py
python3 build_notebook.py
jupyter nbconvert --to notebook --execute --inplace BinPacking_Cohesive.ipynb
latexmk -cd -xelatex report/main.tex
latexmk -cd -xelatex PresentationSlides/Final_Presentation/main_presentation.tex
```

## Single source of truth

- `cohesive_bin_packing.py`: contraction, exact ILP, FFD, randomized initialization, Tabu Search, and validation.
- `run_experiments_final.py`: all 17 main experiments and generated LaTeX tables.
- `run_robustness.py`: 30 independent randomized starts per medium instance.
- `generate_report_artifacts.py`: all report/presentation charts from CSV data.
- `tests/test_cohesive_bin_packing.py`: automated correctness tests.

## Interpretation of reported gaps

For instances with at most 30 original items, the reference is a CBC-proven ILP optimum. For larger instances, the reference is a valid lower bound. Reaching that lower bound proves optimality; a positive gap to the lower bound is not necessarily the true optimality gap.

The `hard_*` instances are synthetic difficult mixtures inspired by bin-packing literature; they are not official Falkenauer benchmark instances.

"""Core algorithms for cohesive one-dimensional bin packing.

Hard must-link constraints are contracted into connected components.  The
resulting instance is exactly a classical one-dimensional bin-packing problem
on component weights; every function in this module works on that contracted
representation and maps solutions back to original items when needed.
"""

from __future__ import annotations

import math
import random
import time
from collections.abc import Iterable, Sequence
from dataclasses import dataclass
from pathlib import Path
from typing import Any, TypedDict, cast

import networkx as nx
from pulp import (
    LpBinary,
    LpMinimize,
    LpProblem,
    LpStatusOptimal,
    LpVariable,
    PULP_CBC_CMD,
    lpSum,
    value,
)

Move = tuple[Any, ...]


class Instance(TypedDict):
    n: int
    C: int
    weights: list[int]
    pairs: list[tuple[int, int]]


@dataclass(frozen=True)
class ContractedInstance:
    capacity: int
    components: tuple[tuple[int, ...], ...]
    weights: tuple[int, ...]


@dataclass
class SearchResult:
    num_bins: int | None
    bins: list[list[int]] | None
    components: tuple[tuple[int, ...], ...]
    component_weights: tuple[int, ...]
    history: list[int]
    solve_time: float
    initial_bins: int | None
    iterations: int
    seed: int | None
    valid: bool


def load_instance(filepath: str | Path) -> Instance:
    data: dict[str, Any] = {}
    with Path(filepath).open("r", encoding="utf-8") as handle:
        for raw_line in handle:
            line = raw_line.strip()
            if not line:
                continue
            key, raw_value = line.split("=", 1)
            if key in {"n", "C"}:
                data[key] = int(raw_value)
            elif key == "weights":
                data[key] = [int(v) for v in raw_value.split(",") if v]
            elif key == "pairs":
                data[key] = [tuple(map(int, pair.split("-"))) for pair in raw_value.split(",") if pair]

    required = {"n", "C", "weights", "pairs"}
    missing = required.difference(data)
    if missing:
        raise ValueError(f"Missing fields in {filepath}: {sorted(missing)}")
    if data["n"] != len(data["weights"]):
        raise ValueError(f"Filename/data instance has n={data['n']} but {len(data['weights'])} weights")
    if data["C"] <= 0 or any(weight <= 0 for weight in data["weights"]):
        raise ValueError("Capacity and item weights must be positive")
    for i, j in data["pairs"]:
        if not (0 <= i < data["n"] and 0 <= j < data["n"] and i != j):
            raise ValueError(f"Invalid cohesion edge {(i, j)}")
    return cast(Instance, cast(object, data))


def contract_instance(instance: Instance) -> ContractedInstance:
    graph = nx.Graph()
    graph.add_nodes_from(range(instance["n"]))
    graph.add_edges_from(instance["pairs"])
    components = tuple(
        sorted((tuple(sorted(component)) for component in nx.connected_components(graph)), key=lambda c: c[0])
    )
    weights = tuple(sum(instance["weights"][item] for item in component) for component in components)
    return ContractedInstance(instance["C"], components, weights)


def is_feasible_contracted(contracted: ContractedInstance) -> bool:
    return all(weight <= contracted.capacity for weight in contracted.weights)


def lower_bound(component_weights: Sequence[int], capacity: int) -> int:
    """A valid lower bound combining volume and items heavier than C/2."""
    if capacity <= 0:
        raise ValueError("Capacity must be positive")
    volume = math.ceil(sum(component_weights) / capacity)
    large_items = sum(weight > capacity / 2 for weight in component_weights)
    return max(volume, large_items)


def first_fit_decreasing(component_weights: Sequence[int], capacity: int) -> list[list[int]]:
    order = sorted(range(len(component_weights)), key=lambda cid: (-component_weights[cid], cid))
    bins: list[list[int]] = []
    loads: list[int] = []
    for cid in order:
        weight = component_weights[cid]
        for bin_index, load in enumerate(loads):
            if load + weight <= capacity:
                bins[bin_index].append(cid)
                loads[bin_index] += weight
                break
        else:
            bins.append([cid])
            loads.append(weight)
    return bins


def randomized_first_fit(
    component_weights: Sequence[int], capacity: int, seed: int
) -> list[list[int]]:
    rng = random.Random(seed)
    order = list(range(len(component_weights)))
    rng.shuffle(order)
    bins: list[list[int]] = []
    loads: list[int] = []
    for cid in order:
        weight = component_weights[cid]
        feasible_bins = [index for index, load in enumerate(loads) if load + weight <= capacity]
        if feasible_bins:
            destination = rng.choice(feasible_bins)
            bins[destination].append(cid)
            loads[destination] += weight
        else:
            bins.append([cid])
            loads.append(weight)
    return bins


def bin_loads(bins: Sequence[Sequence[int]], component_weights: Sequence[int]) -> list[int]:
    return [sum(component_weights[cid] for cid in bin_) for bin_ in bins]


def validate_component_solution(
    bins: Sequence[Sequence[int]], component_weights: Sequence[int], capacity: int
) -> tuple[bool, list[str]]:
    errors: list[str] = []
    flattened = [cid for bin_ in bins for cid in bin_]
    expected = list(range(len(component_weights)))
    if sorted(flattened) != expected:
        errors.append("Every component must occur exactly once")
    for index, load in enumerate(bin_loads(bins, component_weights)):
        if load > capacity:
            errors.append(f"Bin {index} exceeds capacity: {load}>{capacity}")
    return not errors, errors


def expand_solution(
    bins: Sequence[Sequence[int]], components: Sequence[Sequence[int]]
) -> list[list[int]]:
    return [[item for cid in bin_ for item in components[cid]] for bin_ in bins]


def validate_item_solution(instance: Instance, item_bins: Sequence[Sequence[int]]) -> tuple[bool, list[str]]:
    errors: list[str] = []
    item_to_bin: dict[int, int] = {}
    for bin_index, bin_ in enumerate(item_bins):
        load = sum(instance["weights"][item] for item in bin_)
        if load > instance["C"]:
            errors.append(f"Bin {bin_index} exceeds capacity: {load}>{instance['C']}")
        for item in bin_:
            if item in item_to_bin:
                errors.append(f"Item {item} occurs more than once")
            item_to_bin[item] = bin_index
    if set(item_to_bin) != set(range(instance["n"])):
        errors.append("Every original item must occur exactly once")
    for i, j in instance["pairs"]:
        if item_to_bin.get(i) != item_to_bin.get(j):
            errors.append(f"Cohesion edge {(i, j)} is violated")
    return not errors, errors


def generate_neighborhood(
    bins: Sequence[Sequence[int]], component_weights: Sequence[int], capacity: int, include_eject: bool = True
) -> list[Move]:
    moves: list[Move] = []
    loads = bin_loads(bins, component_weights)
    number_of_bins = len(bins)

    for source, source_bin in enumerate(bins):
        for cid in source_bin:
            weight = component_weights[cid]
            for destination in range(number_of_bins):
                if source != destination and loads[destination] + weight <= capacity:
                    moves.append(("move", cid, source, destination))
            if include_eject and len(source_bin) > 1:
                moves.append(("eject", cid, source))

    for first_bin in range(number_of_bins):
        for second_bin in range(first_bin + 1, number_of_bins):
            for first_cid in bins[first_bin]:
                first_weight = component_weights[first_cid]
                for second_cid in bins[second_bin]:
                    second_weight = component_weights[second_cid]
                    if (
                        loads[first_bin] - first_weight + second_weight <= capacity
                        and loads[second_bin] - second_weight + first_weight <= capacity
                    ):
                        moves.append(("swap", first_cid, first_bin, second_cid, second_bin))
            if loads[first_bin] + loads[second_bin] <= capacity:
                moves.append(("merge", first_bin, second_bin))
    return moves


def move_components(move: Move, bins: Sequence[Sequence[int]]) -> tuple[int, ...]:
    if move[0] in {"move", "eject"}:
        return (move[1],)
    if move[0] == "swap":
        return (move[1], move[3])
    if move[0] == "merge":
        return tuple(bins[move[2]])
    raise ValueError(f"Unknown move: {move}")


def calculate_delta(
    bins: Sequence[Sequence[int]], loads: Sequence[int], component_weights: Sequence[int], move: Move
) -> tuple[int, int]:
    """Return (change in bin count, change in sum of squared loads) in O(1)."""
    kind = move[0]
    if kind == "move":
        _, cid, source, destination = move
        weight = component_weights[cid]
        delta_bins = -1 if len(bins[source]) == 1 else 0
        delta_sq = (
            (loads[source] - weight) ** 2
            + (loads[destination] + weight) ** 2
            - loads[source] ** 2
            - loads[destination] ** 2
        )
        return delta_bins, delta_sq
    if kind == "eject":
        _, cid, source = move
        weight = component_weights[cid]
        return 1, (loads[source] - weight) ** 2 + weight**2 - loads[source] ** 2
    if kind == "swap":
        _, first_cid, first_bin, second_cid, second_bin = move
        first_weight = component_weights[first_cid]
        second_weight = component_weights[second_cid]
        new_first = loads[first_bin] - first_weight + second_weight
        new_second = loads[second_bin] - second_weight + first_weight
        return 0, new_first**2 + new_second**2 - loads[first_bin] ** 2 - loads[second_bin] ** 2
    if kind == "merge":
        _, first_bin, second_bin = move
        return -1, (loads[first_bin] + loads[second_bin]) ** 2 - loads[first_bin] ** 2 - loads[second_bin] ** 2
    raise ValueError(f"Unknown move: {move}")


def apply_move(bins: Sequence[Sequence[int]], move: Move) -> list[list[int]]:
    new_bins = [list(bin_) for bin_ in bins]
    kind = move[0]
    if kind == "move":
        _, cid, source, destination = move
        new_bins[source].remove(cid)
        new_bins[destination].append(cid)
    elif kind == "eject":
        _, cid, source = move
        new_bins[source].remove(cid)
        new_bins.append([cid])
    elif kind == "swap":
        _, first_cid, first_bin, second_cid, second_bin = move
        new_bins[first_bin].remove(first_cid)
        new_bins[second_bin].remove(second_cid)
        new_bins[first_bin].append(second_cid)
        new_bins[second_bin].append(first_cid)
    elif kind == "merge":
        _, first_bin, second_bin = move
        new_bins[first_bin].extend(new_bins[second_bin])
        new_bins[second_bin].clear()
    else:
        raise ValueError(f"Unknown move: {move}")
    return [bin_ for bin_ in new_bins if bin_]


def tabu_search(
    instance: Instance,
    *,
    max_iter: int = 1000,
    tabu_tenure: int = 12,
    seed: int = 42,
    initial: str = "ffd",
    stagnation_limit: int | None = None,
    target: int | None = None,
) -> SearchResult:
    contracted = contract_instance(instance)
    started = time.perf_counter()
    if not is_feasible_contracted(contracted):
        return SearchResult(None, None, contracted.components, contracted.weights, [], 0.0, None, 0, seed, False)

    if initial == "ffd":
        bins = first_fit_decreasing(contracted.weights, contracted.capacity)
    elif initial == "random":
        bins = randomized_first_fit(contracted.weights, contracted.capacity, seed)
    else:
        raise ValueError("initial must be 'ffd' or 'random'")

    rng = random.Random(seed)
    best_bins = [list(bin_) for bin_ in bins]
    best_cost = len(best_bins)
    initial_cost = best_cost
    history = [best_cost]
    tabu_until: dict[int, int] = {}
    no_improvement = 0
    stagnation_limit = stagnation_limit or max(75, max_iter // 8)
    iterations = 0

    if target is not None and best_cost <= target:
        valid, _ = validate_component_solution(best_bins, contracted.weights, contracted.capacity)
        return SearchResult(
            best_cost,
            best_bins,
            contracted.components,
            contracted.weights,
            history,
            round(time.perf_counter() - started, 6),
            initial_cost,
            0,
            seed,
            valid,
        )

    for iteration in range(max_iter):
        iterations = iteration + 1
        loads = bin_loads(bins, contracted.weights)
        neighborhood = generate_neighborhood(bins, contracted.weights, contracted.capacity, include_eject=True)
        if not neighborhood:
            break

        current_cost = len(bins)
        admissible: list[tuple[tuple[int, int, float], Move]] = []
        eject_candidates: list[tuple[tuple[int, int, float], Move]] = []
        for move in neighborhood:
            delta_bins, delta_sq = calculate_delta(bins, loads, contracted.weights, move)
            moved = move_components(move, bins)
            is_tabu = any(tabu_until.get(cid, -1) >= iteration for cid in moved)
            aspiration = current_cost + delta_bins < best_cost
            if is_tabu and not aspiration:
                continue
            tie_break = rng.random()
            candidate = ((delta_bins, -delta_sq, tie_break), move)
            if move[0] == "eject":
                eject_candidates.append(candidate)
            else:
                admissible.append(candidate)

        # Intensify normally.  After sustained stagnation, force one admissible
        # EJECT move (at most best+1 bins) to diversify before continuing.
        force_eject = no_improvement > 0 and no_improvement % max(20, stagnation_limit // 3) == 0
        pool = eject_candidates if force_eject and eject_candidates and current_cost <= best_cost else admissible
        if not pool:
            pool = eject_candidates
        if not pool:
            break

        _, selected = min(pool, key=lambda candidate: candidate[0])
        moved = move_components(selected, bins)
        bins = apply_move(bins, selected)
        tenure = rng.randint(max(3, tabu_tenure - 2), tabu_tenure + 2)
        for cid in moved:
            tabu_until[cid] = iteration + tenure

        current_cost = len(bins)
        history.append(current_cost)
        if current_cost < best_cost:
            best_cost = current_cost
            best_bins = [list(bin_) for bin_ in bins]
            no_improvement = 0
            if target is not None and best_cost <= target:
                break
        else:
            no_improvement += 1
        if no_improvement >= stagnation_limit:
            break

    valid, _ = validate_component_solution(best_bins, contracted.weights, contracted.capacity)
    item_valid, _ = validate_item_solution(instance, expand_solution(best_bins, contracted.components))
    return SearchResult(
        best_cost,
        best_bins,
        contracted.components,
        contracted.weights,
        history,
        round(time.perf_counter() - started, 6),
        initial_cost,
        iterations,
        seed,
        valid and item_valid,
    )


def multi_start_tabu_search(
    instance: Instance,
    *,
    seeds: Iterable[int],
    max_iter: int = 1000,
    tabu_tenure: int = 12,
    target: int | None = None,
) -> tuple[SearchResult, list[SearchResult]]:
    results = [
        tabu_search(
            instance,
            max_iter=max_iter,
            tabu_tenure=tabu_tenure,
            seed=seed,
            initial="random",
            target=target,
        )
        for seed in seeds
    ]
    feasible_results = [result for result in results if result.num_bins is not None and result.valid]
    if not feasible_results:
        raise ValueError("No feasible result was produced")
    best = min(feasible_results, key=lambda result: (result.num_bins, result.solve_time))
    return best, results


def solve_exact_ilp(instance: Instance, *, time_limit: int = 120, msg: bool = False) -> dict[str, Any]:
    """Solve the exact component-level formulation and accept only optimal CBC output."""
    contracted = contract_instance(instance)
    if not is_feasible_contracted(contracted):
        return {"status": "Infeasible", "status_code": -1, "optimal": False, "num_bins": None, "solve_time": 0.0}

    number_of_components = len(contracted.weights)
    candidate_bins = range(number_of_components)
    problem = LpProblem("CohesiveBinPackingContracted", LpMinimize)
    x = {
        (cid, bin_index): LpVariable(f"x_{cid}_{bin_index}", cat=LpBinary)
        for cid in range(number_of_components)
        for bin_index in candidate_bins
    }
    y = {bin_index: LpVariable(f"y_{bin_index}", cat=LpBinary) for bin_index in candidate_bins}
    problem += lpSum(y[bin_index] for bin_index in candidate_bins)
    for cid in range(number_of_components):
        problem += lpSum(x[cid, bin_index] for bin_index in candidate_bins) == 1
    for bin_index in candidate_bins:
        problem += (
            lpSum(contracted.weights[cid] * x[cid, bin_index] for cid in range(number_of_components))
            <= contracted.capacity * y[bin_index]
        )
    for bin_index in range(number_of_components - 1):
        problem += y[bin_index] >= y[bin_index + 1]

    started = time.perf_counter()
    status_code = problem.solve(PULP_CBC_CMD(msg=msg, timeLimit=time_limit, gapRel=0.0))
    elapsed = round(time.perf_counter() - started, 6)
    optimal = status_code == LpStatusOptimal
    objective = cast(float | None, value(problem.objective)) if optimal else None
    return {
        "status": "Optimal" if optimal else "Not optimal",
        "status_code": status_code,
        "optimal": optimal,
        "num_bins": int(round(objective)) if objective is not None else None,
        "solve_time": elapsed,
    }

\begin{abstract}
این پروژه مسئله بسته‌بندی یک‌بعدی در سطل‌ها با قیود چسبندگی سخت
(\lr{must-link}) را بررسی می‌کند. ابتدا یک مدل برنامه‌ریزی خطی عدد صحیح ارائه
می‌شود. سپس با انقباض مؤلفه‌های همبند گراف چسبندگی، مسئله به‌طور دقیق به یک
مسئله کلاسیک \lr{Bin Packing} روی اَبَراقلام تبدیل می‌شود. برای حل نمونه‌های
بزرگ، الگوریتم \lr{First Fit Decreasing} با جستجوی تابو ترکیب شده است. جستجو
از چهار عملگر \lr{MOVE، SWAP، MERGE} و \lr{EJECT}، ارزیابی دلتای ثابت، حافظه
تابو مبتنی بر شناسه مؤلفه، معیار تنفس و دوره تصدی پویا استفاده می‌کند.

پیاده‌سازی واحد پروژه روی ۱۷ نمونه متنی در سه مقیاس و دو خانواده داده اجرا و
تمام جواب‌ها با آزمون مستقل ظرفیت، پوشش اقلام و قیود چسبندگی اعتبارسنجی شدند.
در نمونه‌های دارای حداکثر ۳۰ قلم، جواب دقیق \lr{ILP} محاسبه شد و روش پیشنهادی
در هر شش نمونه به مقدار بهینه رسید. در تمام نمونه‌های استاندارد، جواب با کران
پایین معتبر برابر بود. در نمونه‌های مصنوعی دشوار ۳۰۰ و ۵۰۰ قلمی، جستجوی تابو
جواب \lr{FFD} را یک سطل بهبود داد و به کران پایین رسید؛ در نمونه دشوار ۱۵۰
قلمی فاصله نسبت به کران پایین ۲٫۰۴ درصد باقی ماند. همچنین ۳۰ شروع تصادفی مستقل

برای نمونه دشوار ۱۰۰ قلمی، میانگین از ۳۶٫۲۰ به ۳۳٫۴۳ سطل رسید.

\section{الگوریتم پیشنهادی}

\subsection{معماری حل}
قیود چسبندگی سخت یک رابطه هم‌مکانی انتقالی ایجاد می‌کنند. بنابراین ابتدا گراف
$G=(I,E)$ ساخته و مؤلفه‌های همبند آن در زمان $O(|I|+|E|)$ استخراج می‌شوند.
هر مؤلفه به یک اَبَرقلم با وزن مجموع اعضا تبدیل می‌شود. اگر وزن یک مؤلفه از
ظرفیت $C$ بیشتر باشد، نمونه ناموجه است؛ در غیر این صورت مسئله منقبض‌شده دقیقاً
یک \lr{BPP} کلاسیک است و هر جواب آن به یک جواب معتبر برای اقلام اصلی بسط
می‌یابد.

\subsection{جواب اولیه}
دو روش مقداردهی اولیه در پیاده‌سازی واحد پروژه وجود دارد:
\begin{itemize}
  \item \textbf{\lr{FFD}:} مؤلفه‌ها به ترتیب نزولی وزن در نخستین سطل ممکن قرار
  می‌گیرند. این روش در آزمایش اصلی استفاده شده است.
  \item \textbf{\lr{Randomized First Fit}:} ترتیب مؤلفه‌ها و انتخاب مقصد ممکن
  با یک \lr{seed} مشخص تصادفی می‌شود. این روش برای ۳۰ اجرای مستقل تحلیل
  پایداری استفاده شده است.
\end{itemize}
هر دو سازنده فقط جواب موجه تولید می‌کنند.

\subsection{ساختار همسایگی}
تمام حرکات روی مؤلفه‌ها، نه اقلام منفرد، انجام می‌شوند:
\begin{itemize}
  \item \textbf{\lr{MOVE}:} انتقال یک مؤلفه به سطل موجود، مشروط به ظرفیت؛
  \item \textbf{\lr{SWAP}:} مبادله دو مؤلفه از دو سطل متفاوت؛
  \item \textbf{\lr{MERGE}:} ادغام دو سطل در صورت کافی بودن ظرفیت؛
  \item \textbf{\lr{EJECT}:} انتقال یک مؤلفه از سطل چندعضوی به سطل جدید برای
  تنوع‌بخشی جستجو.
\end{itemize}
حرکت \lr{EJECT} موقتاً یک سطل اضافه می‌کند؛ ازاین‌رو در جستجوی عادی انتخاب
نمی‌شود. پس از یک دوره رکود، اگر جواب جاری بیش از یک سطل از بهترین جواب فاصله
نداشته باشد، یک \lr{EJECT} مجاز به‌صورت کنترل‌شده اجرا می‌شود تا ساختار محلی
شکسته شود.

\subsection{تابع ارزیابی، حافظه و توقف}
معیار اصلی کمینه‌سازی تعداد سطل‌هاست. در تساوی معیار اصلی، حرکت با بیشترین
افزایش در
$\sum_b L_b^2$
انتخاب می‌شود؛ این معیار ثانویه سطل‌های پرتر و سطل‌های نزدیک به تخلیه را ترجیح
می‌دهد. تابع \texttt{calculate\_delta} تغییر تعداد سطل‌ها و تغییر مجموع مربعات
بار را بدون ساخت جواب کامل و در $O(1)$ محاسبه می‌کند. تنها حرکت منتخب با یک
کپی خطی از ساختار سطل‌ها اعمال می‌شود.

حافظه تابو به‌جای ذخیره tuple حرکت، شناسه مؤلفه‌های جابه‌جا‌شده را نگه می‌دارد؛
بنابراین حرکت معکوس فوری نیز ممنوع می‌شود. برای پارامتر پایه $T$، دوره تصدی هر
حرکت به‌صورت تصادفی از بازه
$[\max(3,T-2),T+2]$
انتخاب می‌شود. اگر حرکت تابو جوابی بهتر از بهترین جواب تاریخی تولید کند، معیار
تنفس محدودیت را لغو می‌کند. الگوریتم با رسیدن به مقدار مرجع معتبر، رسیدن به
حداکثر تکرار یا عبور از حد رکود متوقف می‌شود.

\subsection{شبه‌کد}
\begin{algorithm}[htbp]
\caption{جستجوی تابو برای بسته‌بندی با قیود چسبندگی}\label{alg:tabu}
\KwIn{اقلام $I$، وزن‌ها $w$، ظرفیت $C$، یال‌ها $E$، پارامترهای $MaxIter,T$}
\KwOut{بهترین بسته‌بندی معتبر $B^*$}
$Components \gets ConnectedComponents(I,E)$\;
\If{$\exists c: W_c>C$}{\Return \lr{Infeasible}}
$B \gets FFD(Components)$; $B^*\gets B$; $TabuUntil\gets\emptyset$\;
\For{$t=1$ \KwTo $MaxIter$}{
  $N(B)\gets MOVE\cup SWAP\cup MERGE\cup EJECT$\;
  \ForEach{$m\in N(B)$}{
    $(\Delta_{bins},\Delta_{sq})\gets CalculateDelta(m)$\;
    حرکت‌های تابو را، مگر در صورت برقراری معیار تنفس، حذف کن\;
  }
  \eIf{رکود دوره‌ای و \lr{EJECT} مجاز وجود دارد}{
    $m^*\gets$ بهترین \lr{EJECT} مجاز\;
  }{
    $m^*\gets\arg\min_m(\Delta_{bins},-\Delta_{sq})$\;
  }
  $B\gets ApplyMove(B,m^*)$\;
  $TabuUntil[c]\gets t+Uniform(\max(3,T-2),T+2)$ برای مؤلفه‌های حرکت\;
  \If{$|B|<|B^*|$}{$B^*\gets B$}
  \If{شرط توقف برقرار است}{\textbf{break}}
}
اعتبار ظرفیت، پوشش مؤلفه‌ها و قیود چسبندگی را بررسی کن\;
\Return $B^*$\;
\end{algorithm}

\subsection{پیچیدگی و پیاده‌سازی}
اگر $m$ تعداد مؤلفه‌ها و $b\le m$ تعداد سطل‌ها باشد، تولید حرکت‌های
\lr{MOVE} از مرتبه $O(mb)$، \lr{SWAP} از مرتبه $O(m^2)$، \lr{MERGE} از
مرتبه $O(b^2)$ و \lr{EJECT} از مرتبه $O(m)$ است. بنابراین هر تکرار در بدترین
حالت $O(m^2)$ و کل جستجو $O(MaxIter\,m^2)$ زمان دارد. چون همسایگی به‌صورت
صریح ساخته می‌شود، حافظه بدترین‌حالت نیز $O(m^2)$ است. اعمال حرکت منتخب
$O(m)$ زمان می‌برد، اما تنها یک‌بار در هر تکرار انجام می‌شود.

برای جلوگیری از اختلاف نسخه‌ها، مدل دقیق، الگوریتم ابتکاری، اعتبارسنجی و تمام
آزمایش‌ها از ماژول واحد \texttt{cohesive\_bin\_packing.py} فراخوانی می‌شوند.
پنج آزمون خودکار نیز contraction، تشخیص ناموجه بودن، صحت دلتا، اعتبار جواب و
تطابق یک نمونه واقعی با جواب دقیق را بررسی می‌کنند.

\section{کاربردها و حدود انطباق مدل}

مدل این گزارش یک مدل انتزاعی با سه فرض محدودکننده است: ظرفیت هر سطل یک‌بعدی و یکسان است، روابط چسبندگی قیود سخت و از پیش معلوم‌اند، و تابع هدف فقط تعداد سطل‌ها را کمینه می‌کند. ازاین‌رو، موارد زیر زمینه‌های قابل دفاع برای استفاده از این انتزاع‌اند، نه شواهدی بر انطباق دقیق همه جزئیات عملی با مدل.

\subsection{برش یک‌بعدی مواد (\lr{Cutting Stock})}\label{subsec:lr_cutting_stock}
مسئله برش یک‌بعدی در صنایعی مانند فولاد، کاغذ، شیشه، مبلمان و نساجی رخ می‌دهد و از نظر تخصیص طول‌های موردنیاز به مواد اولیه با ظرفیت محدود، با بسته‌بندی یک‌بعدی ارتباط ساختاری دارد. تنیر و همکاران گونه‌ای صنعتی را بررسی می‌کنند که در آن اقلام می‌توانند تقسیم و سپس با جوش‌کاری دوباره متصل شوند و هدف، هم‌زمان شامل ضایعات و تعداد جوش‌ها است \cite{D1_Tanir2016}. این گونه با مسئله حاضر یکسان نیست: تقسیم‌پذیری و تابع هدف چندمعیاره در مدل ما وجود ندارد و آن مقاله نیز قید \lr{must-link} این گزارش را معرفی نمی‌کند. تنها اگر در یک کاربرد برش، چند قطعه الزاماً از یک شاخه ماده اولیه بریده شوند و سایر جزئیات نادیده گرفته شوند، می‌توان آن گروه را به‌صورت یک مؤلفه چسبندگی مدل کرد.

\subsection{زمان‌بندی کانتینرها و گردش‌کارها (\lr{Cloud Orchestration})}\label{subsec:lr_cloud_orchestration}
در سامانه‌های کانتینری، نگاشت وظایف به منابع پردازشی و رعایت ترتیب اجرای گردش‌کار از مسائل اصلی زمان‌بندی است. چارچوب \lr{KubeAdaptor} برای اجرای گردش‌کار روی کوبرنتیز، سازگاری ترتیب زمان‌بندی، مصرف منابع و اشتراک داده از طریق حجم‌های پویا را بررسی می‌کند \cite{C1_Shan2022}. مرور زمان‌بندی کربن‌آگاه نیز نشان می‌دهد که زمان‌بندی کانتینرها می‌تواند اهدافی مانند بهره‌وری منابع، زمان پاسخ، انرژی و انتشار کربن داشته باشد \cite{C4_Yang2025}.

هیچ‌یک از این دو منبع، مسئله خود را دقیقاً به‌صورت \lr{BPP} یک‌بعدی با قیود هم‌مکانی سخت فرمول‌بندی نمی‌کند. نگاشت به مدل حاضر فقط تحت یک ساده‌سازی مشخص معتبر است: هر وظیفه یک تقاضای اسکالر دارد، نودها ظرفیت یکسان دارند، هر گروه تعیین‌شده باید روی یک نود واحد اجرا شود، و هدف کمینه‌کردن تعداد نودهای فعال است. در سامانه واقعی، ظرفیت‌های چندبعدی پردازنده و حافظه، ناهمگنی نودها، زمان، تقدم وظایف، دسترس‌پذیری و اهداف انرژی می‌توانند این هم‌ارزی را از بین ببرند. بنابراین این حوزه یک انگیزه مدل‌سازی مشروط است، نه شاهدی برای ادعای تطابق دقیق یا کاهش تضمین‌شده تأخیر و کربن به‌وسیله مدل حاضر.

\section{مقدمه}
مسئله بسته‌بندی در سطل‌ها (\lr{Bin Packing Problem} یا به اختصار \lr{BPP}) مسئله‌ای کلاسیک در بهینه‌سازی ترکیبیاتی است: مجموعه‌ای از اقلام با وزن‌های مثبت داده می‌شود و باید همه آن‌ها را، بدون تجاوز از ظرفیت ثابت هر سطل، در کمترین تعداد سطل قرار داد.

در این گزارش گونه‌ای از مسئله بررسی می‌شود که علاوه بر قید ظرفیت، قیود هم‌مکانی سخت یا \lr{must-link} دارد. هر رابطه چسبندگی میان دو قلم الزام می‌کند که آن دو در یک سطل قرار گیرند. چون هم‌مکانی یک رابطه تراگذری است، تمام اقلام واقع در یک مؤلفه همبند گراف چسبندگی ناگزیر در یک سطل مشترک قرار می‌گیرند. این تعریف باید از ترجیح نرم برای نزدیک‌بودن اقلام، و نیز از قیود ناسازگاری که هم‌مکانی را منع می‌کنند، تفکیک شود.

زمان‌بندی گردش‌کارهای کانتینری و تخصیص منابع در کوبرنتیز نمونه‌ای از زمینه‌هایی است که در آن نگاشت وظایف به منابع و سازوکارهای اشتراک داده مطرح‌اند \cite{C1_Shan2022}؛ همچنین ادبیات زمان‌بندی کانتینرها اهداف گوناگونی مانند بهره‌وری منابع، زمان پاسخ و ملاحظات کربنی را بررسی می‌کند \cite{C4_Yang2025}. با این حال، این منابع مدل یک‌بعدیِ دقیق این گزارش یا الزام هم‌مکانی ناشی از ترافیک شبکه را اثبات نمی‌کنند. مدل حاضر تنها در ساده‌سازی‌ای قابل استفاده است که منابع همگن و تک‌ظرفیتی باشند، گروه‌های هم‌مکانی از پیش و به‌صورت سخت تعیین شده باشند، و هدف کمینه‌کردن تعداد منابع فعال باشد.

از دید محاسباتی، contraction هر مؤلفه چسبندگی به یک قلم با وزن مجموع، نمونه را دقیقاً به یک \lr{BPP} کلاسیک تبدیل می‌کند. بنابراین این گونه در بدترین حالت قویاً \lr{NP-hard} است، زیرا حالت $E=\emptyset$ همان \lr{BPP} کلاسیک است \cite{GareyJohnson1979}؛ اما در مدل تعریف‌شده الزاماً از \lr{BPP} سخت‌تر نیست، چون پس از یک پیش‌پردازش چندجمله‌ای همان مسئله روی اقلام تجمیع‌شده حل می‌شود. در ادامه، این هم‌ارزی مبنای مدل‌سازی و طراحی الگوریتم قرار می‌گیرد؛ کیفیت جواب‌های الگوریتمی نیز باید با کران یا جواب بهینه سنجیده شود و صرف اجرای محاسباتی، اثبات بهینگی محسوب نمی‌شود.

\section{مدل‌سازی و تعریف مسئله}

\subsection{مدل برنامه‌ریزی خطی عدد صحیح (\lr{ILP})}\label{subsec:lr_ilp}
فرض کنید $I=\{1,\ldots,n\}$ مجموعه اقلام، $w_i>0$ وزن قلم $i$، و $C>0$ ظرفیت هر سطل باشد. گراف بدون جهت $G=(I,E)$ روابط چسبندگی را نمایش می‌دهد؛ یال $\{i,k\}\in E$ یعنی دو قلم باید در یک سطل قرار گیرند. چون هر جواب موجه حداکثر از $n$ سطل استفاده می‌کند، مجموعه سطل‌های نامزد را $J=\{1,\ldots,n\}$ می‌گیریم. متغیر $x_{ij}$ نشان می‌دهد قلم $i$ در سطل $j$ قرار دارد و $y_j$ استفاده‌شدن سطل $j$ را نشان می‌دهد:
\begin{align}
    \min \quad & \sum_{j\in J} y_j \label{eq:obj} \\
    \text{s.t.}\quad
    & \sum_{j\in J} x_{ij}=1 && \forall i\in I, \label{eq:assign} \\
    & \sum_{i\in I} w_i x_{ij}\leq C y_j && \forall j\in J, \label{eq:capacity} \\
    & x_{ij}=x_{kj} && \forall \{i,k\}\in E,\ \forall j\in J, \label{eq:cohesion} \\
    & x_{ij}\in\{0,1\} && \forall i\in I,\ \forall j\in J, \label{eq:x_binary} \\
    & y_j\in\{0,1\} && \forall j\in J. \label{eq:y_binary}
\end{align}
قید \eqref{eq:assign} هر قلم را دقیقاً یک‌بار تخصیص می‌دهد. قید \eqref{eq:capacity} هم ظرفیت را اعمال می‌کند و هم، به علت مثبت‌بودن وزن‌ها، اجازه نمی‌دهد قلمی در سطل با $y_j=0$ قرار گیرد. برای هر یال، قیود \eqref{eq:cohesion} بردار تخصیص دو رأس را برابر می‌کنند؛ در نتیجه برابری در امتداد هر مسیر منتشر می‌شود و همه رأس‌های یک مؤلفه همبند دقیقاً در یک سطل قرار می‌گیرند. بنابراین لازم نیست برای تمام جفت‌رأس‌های هر مؤلفه قید جداگانه افزوده شود.

\subsection{انقباض مؤلفه‌ها و هم‌ارزی دقیق با \lr{BPP}}\label{subsec:lr_component_based}
مؤلفه‌های همبند $G$ را با $\mathcal{C}$ نشان می‌دهیم و برای هر $c\in\mathcal{C}$ یک اَبَرقلم با وزن
\[
    W_c=\sum_{i\in c} w_i
\]
می‌سازیم. این انقباض فقط یک تقریب یا ابتکار نیست، بلکه تبدیل دقیق مسئله است:
\begin{itemize}
    \item در هر بسته‌بندی موجه مسئله اصلی، قیود \lr{must-link} تمام اقلام $c$ را در یک سطل قرار می‌دهند؛ پس جایگزینی آن‌ها با اَبَرقلمی به وزن $W_c$ ظرفیت و تعداد سطل‌ها را تغییر نمی‌دهد.
    \item برعکس، از هر بسته‌بندی موجه اَبَراقلام می‌توان با قرار دادن همه اقلام $i\in c$ در سطل اَبَرقلم $c$، یک بسته‌بندی موجه مسئله اصلی با همان تعداد سطل ساخت.
\end{itemize}
پس مجموعه جواب‌ها، پس از نادیده‌گرفتن تمایز داخلی هر مؤلفه، متناظر است و مقدار بهینه دقیقاً حفظ می‌شود. اگر برای مؤلفه‌ای $W_c>C$ باشد، نمونه ناموجه است؛ در غیر این صورت مسئله حاصل دقیقاً \lr{BPP} کلاسیک روی $|\mathcal{C}|$ قلم است. مؤلفه‌ها و وزن‌هایشان را می‌توان در زمان $O(|I|+|E|)$ محاسبه کرد.

\subsection{پیچیدگی و تقلیل}\label{subsec:lr_np_hardness}
نسخه تصمیم مسئله می‌پرسد آیا بسته‌بندی موجهی در حداکثر $K$ سطل وجود دارد. این مسئله در \lr{NP} است، زیرا با داشتن تخصیص اقلام می‌توان یکتایی تخصیص، ظرفیت‌ها، قیود یال‌ها و تعداد سطل‌ها را در زمان چندجمله‌ای بررسی کرد.

برای اثبات سختی قوی، یک نمونه دلخواه از نسخه تصمیم \lr{BPP} با اعداد صحیح، شامل $(I,(w_i)_{i\in I},C,K)$، را در نظر می‌گیریم. همان اقلام، وزن‌ها، ظرفیت و $K$ را نگه می‌داریم و فقط $E=\emptyset$ قرار می‌دهیم. نوشتن نمونه خروجی در زمانی خطی نسبت به طول ورودی انجام می‌شود و هیچ مقدار عددی تغییر یا بزرگ نمی‌شود. چون با $E=\emptyset$ هیچ قید چسبندگی وجود ندارد، پاسخ نمونه ساخته‌شده «بله» است اگر و تنها اگر پاسخ نمونه اصلی \lr{BPP} «بله» باشد. بنابراین این یک تقلیل چندجمله‌ایِ حفظ‌کننده اعداد است. از آنجا که \lr{BPP} از طریق \lr{3-Partition} قویاً \lr{NP-hard} است \cite{GareyJohnson1979}، مسئله حاضر نیز در بدترین حالت قویاً \lr{NP-hard} است؛ و با عضویت فوق در \lr{NP}، نسخه تصمیم آن قویاً \lr{NP-complete} است.

این نتیجه نباید چنین تفسیر شود که افزودن این نوع قید همواره مسئله را از \lr{BPP} سخت‌تر می‌کند. در واقع دو تبدیل چندجمله‌ای داریم: \lr{BPP} با انتخاب $E=\emptyset$ حالت خاص مسئله حاضر است، و هر نمونه مسئله حاضر با انقباض مؤلفه‌ها به یک \lr{BPP} کلاسیکِ هم‌ارز تبدیل می‌شود. بنابراین، برای همین تعریفِ صرفاً \lr{must-link}، تفاوت اصلی در پیش‌پردازش و اندازه اقلام تجمیع‌شده است، نه ایجاد یک کلاس بسته‌بندی ذاتاً سخت‌تر.

\subsection{مسائل مرتبط و کران‌های پایه}\label{subsec:32}
\begin{itemize}
    \item در \lr{Bin Packing with Conflicts}، یال‌ها هم‌مکانی را منع می‌کنند؛ ازاین‌رو contraction مؤلفه‌ها که در اینجا معتبر است، برای قیود تعارض معتبر نیست \cite{A1_Bacci2017,A2_DoronArad2020}.
    \item \lr{Cutting Stock} یک خانواده نزدیک از مسائل تخصیص تقاضا به مواد اولیه با ظرفیت محدود است، اما تقسیم‌پذیری اقلام، الگوهای برش و تابع هدف آن می‌تواند با مدل حاضر متفاوت باشد \cite{D1_Tanir2016}.
\end{itemize}

پس از contraction، دو کران پایین ابتدایی مستقیماً از \lr{BPP} کلاسیک به دست می‌آیند. اگر
\[
    L_{\mathrm{cap}}=\left\lceil\frac{\sum_{c\in\mathcal{C}}W_c}{C}\right\rceil
    \quad\text{و}\quad
    L_{>1/2}=\left|\{c\in\mathcal{C}:W_c>C/2\}\right|,
\]
آنگاه $\max\{L_{\mathrm{cap}},L_{>1/2}\}$ کران پایین معتبر است: اولی از مجموع ظرفیت لازم و دومی از این واقعیت ناشی می‌شود که دو اَبَرقلم با وزن بیش از $C/2$ در یک سطل جا نمی‌شوند. این کران‌ها لزوماً مقدار بهینه را تعیین نمی‌کنند و برابر دانستن آن‌ها با یک آزادسازی خطی قوی‌تر، بدون ارائه فرمول‌بندی و اثبات جداگانه، موجه نیست.

"""Reproducible experiment pipeline for the final report.

Every reported number is generated from ``cohesive_bin_packing.py``.  The
script evaluates all standard and synthetic-hard text instances, validates
every heuristic solution, solves n<=30 instances exactly, and writes both CSV
and LaTeX tables.
"""

from __future__ import annotations

import csv
import json
from pathlib import Path

from cohesive_bin_packing import (
    contract_instance,
    first_fit_decreasing,
    load_instance,
    lower_bound,
    solve_exact_ilp,
    tabu_search,
)

ROOT = Path(__file__).resolve().parent
INSTANCES_DIR = ROOT / "instances"
RESULTS_DIR = ROOT / "results"
RESULTS_DIR.mkdir(exist_ok=True)

GROUPS = (
    ("small", "standard", 500, 7),
    ("medium", "standard", 1000, 12),
    ("large", "standard", 1500, 15),
    ("hard_small", "synthetic-hard", 800, 10),
    ("hard_medium", "synthetic-hard", 1200, 14),
    ("hard_large", "synthetic-hard", 1500, 16),
)

FIELDNAMES = [
    "instance",
    "group",
    "instance_type",
    "n",
    "capacity",
    "cohesion_edges",
    "components",
    "lower_bound",
    "exact_bins",
    "exact_status",
    "exact_time_sec",
    "reference_value",
    "reference_type",
    "ffd_bins",
    "tabu_bins",
    "improvement_bins",
    "gap_to_reference_pct",
    "tabu_time_sec",
    "iterations",
    "valid",
]


def latex_escape(text: str) -> str:
    return text.replace("_", r"\_")


def fmt(value: object, digits: int = 3) -> str:
    if value in {None, ""}:
        return "--"
    if isinstance(value, float):
        return f"{value:.{digits}f}"
    return str(value)


def run_all() -> tuple[list[dict], dict[str, list[int]]]:
    rows: list[dict] = []
    histories: dict[str, list[int]] = {}

    for group, instance_type, max_iter, tenure in GROUPS:
        for path in sorted((INSTANCES_DIR / group).glob("*.txt"), key=lambda p: load_instance(p)["n"]):
            instance = load_instance(path)
            contracted = contract_instance(instance)
            bound = lower_bound(contracted.weights, contracted.capacity)
            ffd_bins = len(first_fit_decreasing(contracted.weights, contracted.capacity))

            exact = {
                "num_bins": None,
                "status": "Not run (n>30)",
                "solve_time": None,
                "optimal": False,
            }
            if instance["n"] <= 30:
                exact = solve_exact_ilp(instance, time_limit=120)

            reference_value = exact["num_bins"] if exact["optimal"] else bound
            reference_type = "ILP optimum" if exact["optimal"] else "lower bound"
            search = tabu_search(
                instance,
                max_iter=max_iter,
                tabu_tenure=tenure,
                seed=42,
                initial="ffd",
                target=reference_value,
            )
            if search.num_bins is None:
                raise RuntimeError(f"No feasible solution for {path}")
            gap = 100.0 * (search.num_bins - reference_value) / max(reference_value, 1)
            row = {
                "instance": path.name,
                "group": group,
                "instance_type": instance_type,
                "n": instance["n"],
                "capacity": instance["C"],
                "cohesion_edges": len(instance["pairs"]),
                "components": len(contracted.components),
                "lower_bound": bound,
                "exact_bins": exact["num_bins"] if exact["optimal"] else "",
                "exact_status": exact["status"],
                "exact_time_sec": exact["solve_time"] if exact["optimal"] else "",
                "reference_value": reference_value,
                "reference_type": reference_type,
                "ffd_bins": ffd_bins,
                "tabu_bins": search.num_bins,
                "improvement_bins": ffd_bins - search.num_bins,
                "gap_to_reference_pct": round(gap, 2),
                "tabu_time_sec": search.solve_time,
                "iterations": search.iterations,
                "valid": search.valid,
            }
            if not row["valid"]:
                raise RuntimeError(f"Invalid heuristic solution for {path}")
            rows.append(row)
            histories[path.name] = search.history
            print(
                f"{group:12s} n={instance['n']:4d} LB={bound:3d} FFD={ffd_bins:3d} "
                f"Tabu={search.num_bins:3d} ref={reference_value:3d} "
                f"gap={gap:5.2f}% time={search.solve_time:.4f}s"
            )
    return rows, histories


def write_csv(rows: list[dict]) -> None:
    with (RESULTS_DIR / "results_final.csv").open("w", newline="", encoding="utf-8") as handle:
        writer = csv.DictWriter(handle, fieldnames=FIELDNAMES)
        writer.writeheader()
        writer.writerows(rows)


def write_latex_table(rows: list[dict], *, hard: bool) -> None:
    selected = [row for row in rows if (row["instance_type"] == "synthetic-hard") == hard]
    selected.sort(key=lambda row: row["n"])
    output = RESULTS_DIR / ("hard_results_table.tex" if hard else "results_table.tex")
    caption = (
        "نتایج نمونه‌های مصنوعی دشوار؛ نوع مقدار مرجع در ستون «مرجع» مشخص شده است."
        if hard
        else "نتایج نمونه‌های استاندارد؛ نوع مقدار مرجع در ستون «مرجع» مشخص شده است."
    )
    label = "tab:hard-results" if hard else "tab:standard-results"
    lines = [
        r"\begin{table}[htbp]",
        r"\centering",
        r"\scriptsize",
        r"\begin{tabular}{r l r r r r l r r}",
        r"\toprule",
        r"ردیف & نمونه & $n$ & \lr{FFD} & \lr{Tabu} & مقدار مرجع & مرجع & شکاف (\%) & زمان (s) \\",
        r"\midrule",
    ]
    for index, row in enumerate(selected, 1):
        reference_type = r"\lr{ILP}" if row["reference_type"] == "ILP optimum" else "کران پایین"
        lines.append(
            f"{index} & \\lr{{{latex_escape(row['instance'])}}} & {row['n']} & {row['ffd_bins']} & "
            f"{row['tabu_bins']} & {row['reference_value']} & {reference_type} & "
            f"{row['gap_to_reference_pct']:.2f} & {row['tabu_time_sec']:.4f} \\\\"
        )
    lines.extend(
        [
            r"\bottomrule",
            r"\end{tabular}",
            f"\\caption{{{caption}}}",
            f"\\label{{{label}}}",
            r"\end{table}",
            "",
        ]
    )
    output.write_text("\n".join(lines), encoding="utf-8")


def main() -> None:
    rows, histories = run_all()
    write_csv(rows)
    write_latex_table(rows, hard=False)
    write_latex_table(rows, hard=True)
    (RESULTS_DIR / "histories.json").write_text(
        json.dumps(histories, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    print(f"\nWrote {len(rows)} validated rows to {RESULTS_DIR / 'results_final.csv'}")


if __name__ == "__main__":
    main()

class CohesiveBinPackingTests(unittest.TestCase):
    instance: Instance = {"n": 0, "C": 1, "weights": [], "pairs": []}

    @override
    def setUp(self) -> None:
        self.instance = {
            "n": 6,
            "C": 10,
            "weights": [2, 3, 4, 1, 5, 5],
            "pairs": [(0, 1), (2, 3)],
        }

    def test_contraction_is_exact_and_transitive(self):
        contracted = contract_instance(self.instance)
        self.assertEqual(contracted.components, ((0, 1), (2, 3), (4,), (5,)))
        self.assertEqual(contracted.weights, (5, 5, 5, 5))

    def test_lower_bound_and_ffd(self):
        contracted = contract_instance(self.instance)
        self.assertEqual(lower_bound(contracted.weights, contracted.capacity), 2)
        bins = first_fit_decreasing(contracted.weights, contracted.capacity)
        self.assertEqual(len(bins), 2)
        self.assertTrue(validate_component_solution(bins, contracted.weights, contracted.capacity)[0])
        self.assertTrue(validate_item_solution(self.instance, expand_solution(bins, contracted.components))[0])

    def test_delta_matches_applied_move(self):
        weights = (4, 3, 3, 2)
        capacity = 7
        bins = [[0, 1], [2, 3]]
        loads = bin_loads(bins, weights)
        for move in generate_neighborhood(bins, weights, capacity, include_eject=True):
            before_count = len(bins)
            before_sq = sum(load * load for load in loads)
            after = apply_move(bins, move)
            after_loads = bin_loads(after, weights)
            expected = (len(after) - before_count, sum(load * load for load in after_loads) - before_sq)
            self.assertEqual(calculate_delta(bins, loads, weights, move), expected, move)

    def test_infeasible_component_is_rejected(self):
