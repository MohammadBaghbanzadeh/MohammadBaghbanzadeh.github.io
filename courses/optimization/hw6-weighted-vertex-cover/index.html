# پیاده‌سازی، مدل‌سازی و تحلیل مسئله پوشش رأس وزن‌دار (Weighted Vertex Cover)
## درس: بهینه‌سازی ترکیبیاتی - تمرین ۶ (HW6)

در این دفترچه، مسئله **پوشش رأس وزن‌دار (Weighted Vertex Cover - WVC)** را در ابعاد بزرگ (بین ۱۰۰ تا ۵۰۰ گره) پیاده‌سازی، حل و تحلیل می‌کنیم. این پروژه شامل بخش‌های زیر است:

1. **تعریف مسئله و تولید داده:** تولید گراف‌های تصادفی با وزن‌های مثبت برای رأس‌ها.
2. **مدل‌سازی و حل با برنامه‌ریزی اعداد صحیح (IP):** فرمول‌بندی مسئله در PuLP و حل بهینه گسسته.
3. **حل با آزادسازی خطی (LP Relaxation):**
   - **حالت اول:** با قید $0 \le x_i \le 1$.
   - **حالت دوم:** با قید $x_i \ge 0$ (بدون حد بالا) برای بررسی اینکه آیا جواب بهینه همچنان $\le 1$ می‌ماند یا خیر.
4. **گرد کردن (Rounding):** اعمال مکانیسم گرد کردن روی جواب LP برای ساخت یک جواب شدنی گسسته و بررسی قضیه ضریب تقریب ۲.
5. **جستجوی محلی (Local Search):** طراحی الگوریتم جستجوی محلی (Hill Climbing با بهبود ۱-swap و حذف افزونگی) برای بهبود جواب گرد شده.
6. **آزمایش‌های عددی و تحلیل نتایج (Gap & Runtime):** ارزیابی کارایی، تحلیل شکاف IP-LP، زمان اجرا و نسبت تقریب در ابعاد بزرگ.


## ۱. تعریف مسئله و تولید داده

در نظریه گراف، **پوشش رأس (Vertex Cover)** در یک گراف غیرجهت‌دار $G = (V, E)$، مجموعه‌ای از رأس‌ها $C \subseteq V$ است به‌طوری که برای هر یال $(u, v) \in E$، حداقل یکی از دو رأس $u$ یا $v$ در $C$ باشد (یعنی یال توسط مجموعه رأس‌ها پوشش داده شود).

در **مسئله پوشش رأس وزن‌دار (Weighted Vertex Cover)**، به هر رأس $i \in V$ یک وزن مثبت $w_i > 0$ اختصاص داده می‌شود. هدف کمینه‌سازی مجموع وزن رأس‌های موجود در پوشش است:
$$ \min \sum_{i \in V} w_i x_i $$

در این بخش، کدی برای تولید گراف تصادفی با وزن‌های تصادفی برای رأس‌ها می‌نویسیم. برای تولید گراف‌ها از مدل Erdős-Rényi استفاده می‌کنیم.

def generate_weighted_graph(n, p=0.05, weight_range=(10, 100)):
    """
    تولید یک گراف تصادفی غیرجهت‌دار بر اساس مدل Erdős-Rényi و انتساب وزن‌های مثبت به گره‌ها
    """
    G = nx.erdos_renyi_graph(n, p)
    
    # انتساب وزن‌های مثبت تصادفی به گره‌ها
    for u in G.nodes():
        G.nodes[u]['weight'] = random.randint(weight_range[0], weight_range[1])
        
    return G

# تولید یک نمونه کوچک برای نمایش بصری و صحت‌سنجی اولیه
small_G = generate_weighted_graph(n=15, p=0.25)
print("تعداد گره‌ها:", small_G.number_of_nodes())
print("تعداد یال‌ها:", small_G.number_of_edges())
print("وزن ۵ گره اول:")
for u in list(small_G.nodes())[:5]:
    print(f"گره {u}: وزن {small_G.nodes[u]['weight']}")

def solve_wvc_ip(G):
    """
    حل مسئله پوشش رأس وزن‌دار با استفاده از برنامه‌ریزی اعداد صحیح (IP)
    """
    start_time = time.time()
    
    # تعریف مسئله کمینه‌سازی
    prob = pulp.LpProblem("Weighted_Vertex_Cover_IP", pulp.LpMinimize)
    
    # تعریف متغیرهای تصمیم‌گیری (باینری 0 و 1)
    x = pulp.LpVariable.dicts("x", G.nodes(), cat=pulp.LpBinary)
    
    # تعریف تابع هدف: مجموع وزن گره‌های انتخاب شده
    prob += pulp.lpSum(G.nodes[i]['weight'] * x[i] for i in G.nodes())
    
    # قیدها: برای هر یال (i, j) حداقل یکی از دو سر یال باید در پوشش باشد
    for u, v in G.edges():
        prob += x[u] + x[v] >= 1
        
    # حل مسئله با خاموش کردن خروجی متنی سالور و محدودیت زمانی ۱۵ ثانیه‌ای
    prob.solve(pulp.PULP_CBC_CMD(msg=False, timeLimit=15))
    
    end_time = time.time()
    
    # استخراج جواب
    solution = {i: int(x[i].varValue) for i in G.nodes()}
    objective_val = pulp.value(prob.objective)
    
    return objective_val, solution, end_time - start_time

# تست تابع روی گراف کوچک
ip_val, ip_sol, ip_time = solve_wvc_ip(small_G)

def solve_wvc_lp_case1(G):
    """
    حل آزادسازی خطی با قید 0 <= x_i <= 1
    """
    start_time = time.time()
    
    prob = pulp.LpProblem("Weighted_Vertex_Cover_LP_Case1", pulp.LpMinimize)
    x = pulp.LpVariable.dicts("x", G.nodes(), lowBound=0, upBound=1, cat=pulp.LpContinuous)
    
    # تابع هدف
    prob += pulp.lpSum(G.nodes[i]['weight'] * x[i] for i in G.nodes())
    
    # قیدها
    for u, v in G.edges():
        prob += x[u] + x[v] >= 1
        
    prob.solve(pulp.PULP_CBC_CMD(msg=False))
    end_time = time.time()
    
    solution = {i: x[i].varValue for i in G.nodes()}
    objective_val = pulp.value(prob.objective)
    
    return objective_val, solution, end_time - start_time

def solve_wvc_lp_case2(G):
    """
    حل آزادسازی خطی با قید x_i >= 0 (بدون سقف بالا)
    """
    start_time = time.time()
    
    prob = pulp.LpProblem("Weighted_Vertex_Cover_LP_Case2", pulp.LpMinimize)
    x = pulp.LpVariable.dicts("x", G.nodes(), lowBound=0, cat=pulp.LpContinuous)
    
    # تابع هدف
    prob += pulp.lpSum(G.nodes[i]['weight'] * x[i] for i in G.nodes())
    
    # قیدها
    for u, v in G.edges():
        prob += x[u] + x[v] >= 1
        
    prob.solve(pulp.PULP_CBC_CMD(msg=False))
    end_time = time.time()
    
    solution = {i: x[i].varValue for i in G.nodes()}
    objective_val = pulp.value(prob.objective)
    
    return objective_val, solution, end_time - start_time

# اجرای هر دو حالت روی گراف کوچک و مقایسه مقادیر هدف و زمان اجرا
lp1_val, lp1_sol, lp1_time = solve_wvc_lp_case1(small_G)
lp2_val, lp2_sol, lp2_time = solve_wvc_lp_case2(small_G)

print(f"مقدار هدف LP حالت اول (0 <= x_i <= 1): {lp1_val:.4f}")
print(f"مقدار هدف LP حالت دوم (x_i >= 0): {lp2_val:.4f}")
print(f"اختلاف دو جواب: {abs(lp1_val - lp2_val):.6f}")

max_val_case2 = max(lp2_sol.values())
print(f"بزرگترین مقدار متغیر در جواب بهینه حالت دوم: {max_val_case2:.4f}")

def round_lp_solution(G, lp_solution):
    """
    اعمال مکانیسم گرد کردن روی خروجی LP:
    اگر مقدار متغیر بزرگتر یا مساوی 0.5 بود 1، در غیر این صورت 0.
    """
    rounded_solution = {}
    for i in G.nodes():
        if lp_solution[i] >= 0.5:
            rounded_solution[i] = 1
        else:
            rounded_solution[i] = 0
            
    # محاسبه وزن کل جواب گرد شده
    total_weight = sum(G.nodes[i]['weight'] * rounded_solution[i] for i in G.nodes())
    
    # بررسی شدنی بودن (آیا تمام یال‌ها پوشش داده شده‌اند؟)
    is_feasible = True
    for u, v in G.edges():
        if rounded_solution[u] + rounded_solution[v] < 1:
            is_feasible = False
            break
            
    return total_weight, rounded_solution, is_feasible

# اجرای گرد کردن روی جواب گراف کوچک
round_weight, round_sol, round_feasible = round_lp_solution(small_G, lp1_sol)
print(f"وزن پوشش رأس حاصل از گرد کردن: {round_weight}")
print(f"آیا جواب شدنی است؟ {'بله' if round_feasible else 'خیر'}")
print(f"نسبت جواب گرد شده به بهینه IP: {round_weight / ip_val:.4f}")

def local_search_wvc(G, initial_cover):
    """
    بهبود جواب پوشش رأس با جستجوی محلی (Hill Climbing)
    """
    start_time = time.time()
    
    # کپی کردن پوشش رأس اولیه به صورت مجموعه‌ای از گره‌ها
    cover = {u for u, val in initial_cover.items() if val == 1}
    
    def remove_redundancies(curr_cover):
        # حذف گره‌هایی که تمام همسایگانشان در پوشش هستند
        # کاندیداها را بر اساس وزن نزولی مرتب می‌کنیم تا اول گره‌های سنگین‌تر حذف شوند
        redundant_candidates = sorted(list(curr_cover), key=lambda x: G.nodes[x]['weight'], reverse=True)
        modified = False
        for u in redundant_candidates:
            if all(neighbor in curr_cover for neighbor in G.neighbors(u)):
                curr_cover.remove(u)
                modified = True
        return curr_cover, modified

    # اولین پاکسازی افزونگی‌ها
    cover, _ = remove_redundancies(cover)
    
    improved = True
    iterations = 0
    max_iterations = 1000
    
    while improved and iterations < max_iterations:
        improved = False
        iterations += 1
        
        # مرتب‌سازی گره‌های داخل پوشش برای انتخاب به ترتیب نزولی وزن
        cover_sorted = sorted(list(cover), key=lambda x: G.nodes[x]['weight'], reverse=True)
        
        for u in cover_sorted:
            # پیدا کردن همسایگانی از u که در پوشش نیستند
            uncovered_neighbors = [n for n in G.neighbors(u) if n not in cover]
            
            if len(uncovered_neighbors) == 0:
                cover.remove(u)
                cover, _ = remove_redundancies(cover)
                improved = True
                break
                
            elif len(uncovered_neighbors) == 1:
                v = uncovered_neighbors[0]
                # شرط تعویض: وزن گره خارج از پوشش کمتر از گره داخل پوشش باشد
                if G.nodes[v]['weight'] < G.nodes[u]['weight']:
                    cover.remove(u)
                    cover.add(v)
                    cover, _ = remove_redundancies(cover)
                    improved = True
                    break
            else:
                pass
                
    end_time = time.time()
    
    # ساخت دیکشنری جواب نهایی
    final_cover = {u: (1 if u in cover else 0) for u in G.nodes()}
    final_weight = sum(G.nodes[u]['weight'] for u in cover)
    
    return final_weight, final_cover, end_time - start_time

# تست جستجوی محلی روی گراف کوچک

    # ۴. حل به روش آزادسازی خطی LP (حالت دوم) برای صحت‌سنجی تجربی فرضیه عدم تجاوز متغیرها از ۱
    _, lp2_sol, _ = solve_wvc_lp_case2(G)

        'round_ratio': round_ratio,
        'ls_ratio': ls_ratio,
        'round_feasible': round_feasible,
        'violating_vars_case2': violating_vars
    })
    
    print(f"{n:<8}{num_edges:<8}{ip_weight:<12.1f}{lp_weight:<12.2f}{round_weight:<10.1f}{ls_weight:<10.1f}{ip_time:<12.4f}{lp_time:<12.4f}{ls_time:<12.4f}")

\section{سناریوی پیشنهادی برای ارائه گام‌به‌گام به استاد}

در زمان ارائه نوت‌بوک، بهتر است کدهای هر سلول را همراه با «چرایی» و «دلیل علمی» آن به استاد نشان دهید. از سناریوی زیر استفاده کنید:

\subsection{گام اول: تولید داده تصادفی گراف}
توضیح دهید که یک گراف تصادفی غیرجهت‌دار بر اساس مدل \lr{Erdős-Rényi} با چگالی یال $p=0.05$ تولید کرده و به هر گره یک وزن تصادفی بین ۱۰ تا ۱۰۰ اختصاص داده‌اید.
\begin{itemize}
    \item \textbf{چرا چگالی $0.05$؟} زیرا این مقدار تضمین می‌کند که گراف هم به اندازه کافی جاده (یال) برای پوشش دادن دارد و هم بیش از حد شلوغ نیست که مسئله بدیهی شود.
    \item \textbf{چرا وزن‌های مثبت؟} زیرا اگر وزن‌ها منفی یا صفر باشند، مسئله پوشش رأس وزن‌دار معنای فیزیکی و هزینه‌ای خود را از دست می‌دهد و گره‌های با وزن منفی همواره بدون هزینه انتخاب می‌شوند.
\end{itemize}

\subsection{گام دوم: حل با مدل برنامه‌ریزی اعداد صحیح باینری \lr{(IP)}}
با استفاده از کتابخانه \lr{PuLP} مدل را تعریف کرده و متغیرهای تصمیم‌گیری را به صورت باینری صفر و یک تعریف کرده‌اید.
\begin{tipBox}{محدودیت زمان حل \lr{IP}}
حتماً به استاد بگویید: «در زمان فراخوانی سالور \lr{CBC}، پارامتر محدودیت زمانی را روی ۱۵ ثانیه تنظیم کرده‌ام (\lr{timeLimit=15})؛ زیرا در ابعاد بالای ۱۵۰ رأس، مسئله به دلیل نمایی بودن فضای جستجوی درخت شاخه و حد ($2^N$) بسیار زمان‌بر می‌شود. این محدودیت زمانی باعث می‌شود سالور پس از ۱۵ ثانیه بهترین جواب شدنی پیدا شده را برگرداند تا روند اجرای کل پروژه متوقف نشود.»
\end{tipBox}

\subsection{گام سوم: آزادسازی خطی \lr{(LP Relaxation)} و بررسی حالت بدون سقف}
آزادسازی خطی را یک بار با قید استاندارد $0 \le x_i \le 1$ و بار دیگر با قید $x_i \ge 0$ (بدون حد بالای ۱) اجرا کرده‌اید.
\begin{itemize}
    \item \textbf{توضیح برای استاد:} به استاد نشان دهید که جواب بهینه هر دو مدل کاملاً یکسان است و بزرگترین مقدار متغیر در حالت بدون سقف نیز هرگز از ۱ تجاوز نکرده است.
    \item \textbf{علت تئوری:} چون وزن گره‌ها مثبت است ($w_i > 0$) و ما در حال کمینه کردن وزن کل هستیم، اگر در نقطه بهینه متغیر گرهی از ۱ بزرگتر باشد ($x_u^* > 1$)، سالور می‌تواند بدون نقض قیود یال‌ها ($1 + x_v^* \ge 1$) مقدار آن را به ۱ کاهش داده و مقدار تابع هدف را بهبود بخشد که این موضوع با فرض بهینه بودن در تناقض است.
\end{itemize}

\subsection{گام چهارم: الگوریتم گرد کردن \lr{(Rounding)}}
مقادیر اعشاری خروجی \lr{LP} را با آستانه $0.5$ گرد کرده و یک جواب باینری شدنی ساخته‌اید.
\begin{itemize}
    \item \textbf{اثبات شدنی بودن:} برای هر یال $(u, v)$ قید $x_u^* + x_v^* \ge 1$ برقرار است، پس حداقل یکی از این دو متغیر باید $\ge 0.5$ باشد (طبق اصل لانه کبوتر). بنابراین در زمان گرد کردن، حداقل یکی از آن‌ها به ۱ تبدیل شده و یال به صورت خودکار پوشش داده می‌شود.
    \item \textbf{ضریب تقریب ۲:} اثبات ریاضی نشان می‌دهد وزن جواب گرد شده همواره کمتر یا مساوی ۲ برابر کران پایین \lr{LP} و در نتیجه کمتر یا مساوی ۲ برابر بهینه واقعی \lr{IP} است.
\end{itemize}

\subsection{گام پنجم: بهبود به کمک جستجوی محلی \lr{(Local Search)}}
الگوریتم جستجوی محلی بر پایه روش \lr{Hill Climbing} را اجرا کرده‌اید که دو حرکت اصلی دارد: حذف گره‌های افزونه (که همسایگانشان در پوشش هستند) با ترتیب وزن نزولی، و حرکت تعویضی ۱-\lr{Swap} برای جایگزینی گره‌های سنگین با سبک‌تر.
\begin{itemize}
    \item \textbf{توضیح برای استاد:} جواب گرد شده در گراف‌های تصادفی شلوغ است (چون سالور تمام مقادیر را $0.5$ می‌گیرد و همه به ۱ گرد می‌شوند). جستجوی محلی با هزینه محاسباتی ناچیز (زیر ۱ میلی‌ثانیه)، این افزونگی‌ها را حذف کرده و جواب را به شدت به بهینه واقعی نزدیک می‌کند.
\end{itemize}

\section{آنالیز نتایج تجربی شبیه‌سازی}

استاد بر اساس خروجی‌های جدول آزمایش‌ها ممکن است چند تحلیل بخواهد:

\subsection{پدیده نسبت تقریب دقیقاً ۲.۰ در گرد کردن}
در جدول نتایج ابعاد بزرگ (۱۵۰ رأس به بالا)، متوجه می‌شوید که نسبت جواب گرد شده به \lr{LP} دقیقاً برابر با $2.0$ است.
\begin{itemize}
    \item \textbf{علت:} در گراف‌های تصادفی متقارن با چگالی متوازن و وزن‌های یکنواخت، بهینه‌ترین جواب برای آزادسازی خطی این است که به تمام متغیرها مقدار دقیقاً $0.5$ اختصاص داده شود ($x_i^* = 0.5$). در زمان گرد کردن، طبق قانون آستانه، تمام این مقادیر $0.5$ به ۱ تبدیل می‌شوند؛ یعنی کل گره‌های گراف در پوشش قرار می‌گیرند. این موضوع باعث می‌شود وزن پوشش گرد شده دقیقاً $\sum w_i$ شود که دقیقاً ۲ برابر جواب \lr{LP} (یعنی $0.5 \sum w_i$) است.
\end{itemize}

\subsection{کارایی فوق‌العاده جستجوی محلی}
با وجود اینکه جواب گرد شده بسیار سنگین است و نسبت تقریب ۲ را نشان می‌دهد، اما جستجوی محلی بلافاصله وارد عمل شده و گره‌های اضافی را حذف می‌کند. در جدول خروجی مشخص است که در سایز ۵۰۰ گره، وزن جواب جستجوی محلی با جواب بهینه \lr{IP} کمتر از $0.2\%$ تفاضل دارد، در حالی که زمان اجرای آن نزدیک به صفر است.

\section{سؤالات چالشی استاد در جلسه دفاع و نحوه پاسخ به آن‌ها}

\begin{questionBox}{چرا از ابتدا مستقیماً مدل دقیق \lr{IP} را برای ابعاد بزرگ حل نمی‌کنیم؟}
\textbf{پاسخ پیشنهادی:} زیرا مسئله پوشش رأس وزن‌دار یک مسئله \lr{NP-Hard} است. در ابعاد بزرگ، زمان حل درخت شاخه و حد به صورت نمایی با مرتبه زمانی $2^N$ رشد می‌کند و پیدا کردن جواب بهینه مطلق غیرممکن می‌شود. اما آزادسازی خطی \lr{LP} در زمان چندجمله‌ای (چند میلی‌ثانیه) حل می‌شود و به ما کمک می‌کند با روش گرد کردن و جستجوی محلی، در کسری از ثانیه به جوابی با اختلاف کمتر از ۳٪ نسبت به بهینه واقعی دست یابیم.
\end{questionBox}

\begin{questionBox}{در حذف گره‌های افزونه، چرا کاندیداها را بر اساس وزن نزولی مرتب کردید؟}
\textbf{پاسخ پیشنهادی:} هدف ما کمینه کردن وزن کل پوشش است. حذف یک گره افزونه ممکن است باعث شود همسایگان آن دیگر افزونه نباشند (یعنی نتوانیم آن‌ها را حذف کنیم). بنابراین، با مرتب‌سازی نزولی بر اساس وزن، مطمئن می‌شویم که ابتدا سنگین‌ترین گره‌های افزونه را حذف می‌کنیم تا بیشترین کاهش هزینه در تابع هدف حاصل شود.
\end{questionBox}

\begin{questionBox}{آیا جواب گرد شده همواره شدنی است؟ چطور این را اثبات می‌کنید؟}
\textbf{پاسخ پیشنهادی:} بله، ۱۰۰٪ شدنی است. برای هر یال $(u, v)$ قید $x_u^* + x_v^* \ge 1$ در مدل \lr{LP} ارضا می‌شود. طبق اصل لانه کبوتر، غیرممکن است هر دو متغیر کوچک‌تر از $0.5$ باشند؛ زیرا در این صورت جمعشان کوچک‌تر از ۱ خواهد شد. پس حداقل یکی از آن‌ها $\ge 0.5$ است و طبق آستانه به ۱ گرد می‌شود. در نتیجه، برای تمام یال‌ها حداقل یکی از دو سر یال در پوشش قرار می‌گیرد و جواب شدنی باقی می‌ماند.
\end{questionBox}

\begin{questionBox}{مفهوم شکاف بهینگی \lr{(Optimality Gap)} در آزمایش‌ها چیست؟}
\textbf{پاسخ پیشنهادی:} این شکاف نشان‌دهنده فاصله درصد بین کران پایین (وزن حاصل از \lr{LP Relaxation}) و جواب بهینه گسسته (\lr{IP}) است. هرچه این شکاف کوچک‌تر باشد، یعنی تخمین آزادسازی خطی به جواب واقعی نزدیک‌تر است. در آزمایش‌های تجربی ما این شکاف حول ۳۸٪ به دست آمد که نشان‌دهنده کارکرد طبیعی آزادسازی خطی در تخمین پوشش گره‌ها است.
\end{questionBox}
