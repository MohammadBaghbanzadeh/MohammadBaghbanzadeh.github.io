def fit_and_evaluate(model, X_train, y_train, X_test, y_test, tolerance=0.2):
    # Fit the model
    model.fit(X_train, y_train)
    
    # Make predictions
    y_pred = model.predict(X_test)
    
    # R2 Score
    r2 = r2_score(y_test, y_pred)
    
    # Mean Absolute Percentage Error (MAPE)
    mape= mean_absolute_percentage_error(y_test, y_pred)
    
    #range accuracy
    correct_count = 0
    total_count = len(y_test)

    for true, predicted in zip(y_test, y_pred):
        if abs(true - predicted) <= tolerance*true:
            correct_count += 1

    accuracy = (correct_count / total_count)
    
    #Scatterplot
    plt.scatter(y_test, y_pred)
    plt.show()
    
    return r2, mape , accuracy

import re

def remove_emojis(data):
    emoj = re.compile("["

        u"\U00002702-\U000027B0"
        u"\U000024C2-\U0001F251"
        u"\U0001f926-\U0001f937"
        u"\U00010000-\U0010ffff"
        u"\u2640-\u2642" 
        u"\u2600-\u2B55"
        u"\u200d"
        u"\u23cf"
        u"\u23e9"
        u"\u231a"
        u"\ufe0f"  # dingbats
        u"\u3030"
                      "]+", re.UNICODE)
    return re.sub(emoj, ' ', data)

def remove_parentheses(text):
    return re.sub(r'[()]', ' ',text)

def remove_links(data):
    return re.sub("https?:\\/\\/(?:www\\.)?[-a-zA-Z0-9@:%._\\+~#=]{1,256}\\.[a-zA-Z0-9()]{1,6}\\b(?:[-a-zA-Z0-9()@:%_\\+.~#?&\\/=]*)", ' ', data)

def normalizing_lines(text):          
    return re.sub(r'(\n)+', r' ',text)

def normalizing_punctuations(text):
    text = re.sub('[`~@#$%^&*\-_=+/\\|\[\]<>{}\'«»"]', ' ', text)
    text=text.strip()
    return re.sub(r'([:!?,;،.])+', r' ', text)

def normalize_whitespaces(data):
    data = re.sub("\n+", " ", data)
    data = re.sub("  +", ' ', data)
    data = re.sub("\t+", " ", data)
    data = re.sub("\r+", " ", data)
    data = re.sub("\v+", " ", data)
    data = re.sub("\f+", " ", data)
    return data

    text=remove_emojis(text)
    text=remove_links(text)
    text=remove_parentheses(text)
    text=normalizing_lines(text)
    text=normalize_whitespaces(text)
    text=normalizing_punctuations(text)
    return text

import re

def remove_emojis(data):
    emoj = re.compile("["
        u"\U0001F600-\U0001F64F"  # emoticons
        u"\U0001F300-\U0001F5FF"  # symbols & pictographs
        u"\U0001F680-\U0001F6FF"  # transport & map symbols
        u"\U0001F1E0-\U0001F1FF"  # flags (iOS)
        u"\U00002500-\U00002BEF"  # chinese char
        u"\U00002702-\U000027B0"
        u"\U00002702-\U000027B0"
        u"\U000024C2-\U0001F251"
        u"\U0001f926-\U0001f937"
        u"\U00010000-\U0010ffff"
        u"\u2640-\u2642" 
        u"\u2600-\u2B55"
        u"\u200d"
        u"\u23cf"
        u"\u23e9"
        u"\u231a"
        u"\ufe0f"  # dingbats
        u"\u3030"
                      "]+", re.UNICODE)
    return re.sub(emoj, ' ', data)

def remove_parentheses(text):
    return re.sub(r'[()]', ' ',text)

def remove_links(data):
    return re.sub("https?:\\/\\/(?:www\\.)?[-a-zA-Z0-9@:%._\\+~#=]{1,256}\\.[a-zA-Z0-9()]{1,6}\\b(?:[-a-zA-Z0-9()@:%_\\+.~#?&\\/=]*)", ' ', data)

def normalizing_lines(text):          
    return re.sub(r'(\n)+', r' ',text)

def normalizing_punctuations(text):
    text = re.sub('[`~@#$%^&*\-_=+/\\|\[\]<>{}\'«»"]', ' ', text)
    text=text.strip()
    return re.sub(r'([:!?,;،.])+', r' ', text)

def normalize_whitespaces(data):
    data = re.sub("\n+", " ", data)
    data = re.sub("  +", ' ', data)
    data = re.sub("\t+", " ", data)
    data = re.sub("\r+", " ", data)
    data = re.sub("\v+", " ", data)
    data = re.sub("\f+", " ", data)
    return data

def preprocessing_pipeline(text):
    text=remove_emojis(text)
    text=remove_links(text)
    text=remove_parentheses(text)
    text=normalizing_lines(text)
    text=normalize_whitespaces(text)
    text=normalizing_punctuations(text)
    return text

def fit_and_evaluate(model, X_train, y_train, X_test, y_test, tolerance=0.2):
    # Fit the model
    model.fit(X_train, y_train)
    
    # Make predictions
    y_pred = model.predict(X_test)
    
    # R2 Score
    r2 = r2_score(y_test, y_pred)
    
    # Mean Absolute Percentage Error (MAPE)
    mape= mean_absolute_percentage_error(y_test, y_pred)
    
    #range accuracy
    correct_count = 0
    total_count = len(y_test)

    for true, predicted in zip(y_test, y_pred):
        if abs(true - predicted) <= tolerance*true:
            correct_count += 1

    accuracy = (correct_count / total_count)
    
    #Scatterplot
    plt.scatter(y_test, y_pred)
    plt.show()
    
    return r2, mape , accuracy
